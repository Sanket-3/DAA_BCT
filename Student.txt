pragma solidity ^0.8.18;

contract MaksManagmtSys {

    // Structure
    struct studData {
        uint id;
        string name;
        uint marks;
    }

    // Array
    studData[] public students;

    // Add student
    function addStudent(
        uint _id,
        string memory _name,
        uint _marks
    ) public {
        students.push(studData(_id, _name, _marks));
    }

    // Get student
    function getStudent(uint index)
        public view returns(uint, string memory, uint)
    {
        studData memory s = students[index];

        return (s.id, s.name, s.marks);
    }

    // Total students
    function totalStudents() public view returns(uint) {
        return students.length;
    }

    // Fallback function
    fallback() external {
        revert("Invalid function call");
    }
}
